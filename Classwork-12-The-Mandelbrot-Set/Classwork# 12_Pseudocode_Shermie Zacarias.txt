IMPORT Image processing library

# READ CONFIGURATION 
INITIALIZE empty dictionary called config

OPEN file "config.txt" for reading
READ all lines from file
FOR EACH line IN lines:
    REMOVE leading/trailing whitespaces from line
    SPLIT line by "=" into parameter, value
    
    IF value contains a "." THEN
        STORE value as a floating-point number in config[parameter]
    ELSE
        STORE value as an integer in config[parameter]
    END IF
END FOR
CLOSE file

PRINT config dictionary

# READ CSV DATA 
OPEN file "mandelbrot.csv" for reading
READ all lines into a list called csv_lines
CLOSE file


REMOVE the first item (index 0) from csv_lines

# UNPACK VARIABLES
SET max_iter TO config["max_iter"]
SET width TO config["ancho"]
SET height TO config["alto"]

# CREATE AND PAINT IMAGE 
CREATE new image object 'img' in "HSV" mode with size (width, height)

FOR EACH line IN csv_lines:
    REMOVE leading/trailing whitespaces from line
    SPLIT line by "," into row, column, iterations
    
    CONVERT iterations to integer
    CONVERT row to integer
    CONVERT column to integer
    
    IF iterations EQUALS max_iter THEN
        SET brightness TO 40
    ELSE
        SET brightness TO integer value of ((iterations / max_iter) * 255)
    END IF
    
    # Color format is (Hue/Brightness, Saturation, Value)
    SET PIXEL in 'img' at coordinates (column, row) to color (brightness, 255, 255)
END FOR

# EXPORT IMAGE 
CONVERT 'img' from HSV to RGB format and store in 'img_rgb'
SAVE 'img_rgb' as "mandelbrot.png"

PRINT "DONE"
