BEGIN PROGRAM SchoolManagementSystem

    # Data Structures Initialization
    SET total_users = 8
    SET total_subjects = 7
    
    # Parallel Arrays for user data
    CREATE ARRAY usernames = ["jperez", "dromo", "mjuarez", "mlopez", "euc", "cbalam", "jpedrozo", "dgamboa"]
    CREATE ARRAY passwords = ["1234", "1234", "1234", "1234", "1234", "1234", "1234", "1234"]
    CREATE ARRAY roles = ["student", "student", "student", "student", "student", "student", "professor", "coordinator"]
    CREATE ARRAY names = ["Juan Pérez", "Daniela Romo", "Mauricio Juárez", "María López", "Ernesto Uc", "Carlos Balam", "Jorge Pedrozo", "Didier Gamboa"]
    
    # Array for subjects
    CREATE ARRAY subjects = ["Discrete Mathematics", "Programming", "English II", "Differential Calculus", "Probability and Statistics", "Computer and Server Architecture", "Socio-Emotional Skills and Conflict Management"]
    
    CREATE MATRIX notes[8][7]

    # Login System
    SET logged_in = FALSE
    SET current_role = ""
    SET current_name = ""
    SET user_index = 0
    
    WHILE logged_in is FALSE DO
        PRINT "User: "
        INPUT input_user
        PRINT "Password: "
        INPUT input_pass
        
        SET user_index = 0
        
        # Search for user and validate password
        FOR i = 1 TO total_users DO
            IF usernames[i] EQUALS input_user AND passwords[i] EQUALS input_pass THEN
                SET user_index = i
                SET logged_in = TRUE
                SET current_role = roles[i]
                SET current_name = names[i]
            END IF
        END FOR
        
        IF logged_in is TRUE THEN
            PRINT "Welcome!, " + current_name + " (" + current_role + ")"
        ELSE
            PRINT "Wrong username/password"
        END IF
    END WHILE

    # Role Menus
    IF current_role EQUALS "student" THEN
        PRINT "========================================="
        PRINT "  School Report"
        PRINT "========================================="
        
        SET approved_list = ""
        SET pending_list = ""
        
        FOR j = 1 TO total_subjects DO
            SET current_grade = notes[user_index][j]
            
            // Check if the student passed before printing
            IF current_grade >= 7.0 THEN
                PRINT subjects[j] + ": " + current_grade
                APPEND subjects[j] TO approved_list
            ELSE
                // If they failed, it goes silently to the pending list
                APPEND subjects[j] TO pending_list
            END IF
        END FOR
        
        PRINT "Approved: {" + approved_list + "}"
        PRINT "Pending : {" + pending_list + "}"

    ELSE IF current_role EQUALS "professor" THEN
