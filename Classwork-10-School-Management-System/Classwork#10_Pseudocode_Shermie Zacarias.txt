BEGIN PROGRAM SchoolManagementSystem

    # Data Structures Initialization
    SET total_users = 8
    SET total_subjects = 7
    
    # Parallel Arrays for user data
    CREATE ARRAY usernames = ["jperez", "dromo", "mjuarez", "mlopez", "euc", "cbalam", "jpedrozo", "dgamboa"]
    CREATE ARRAY passwords = ["1234", "1234", "1234", "1234", "1234", "1234", "1234", "1234"]
    CREATE ARRAY roles = ["student", "student", "student", "student", "student", "student", "professor", "coordinator"]
    CREATE ARRAY names = ["Juan Pérez", "Daniela Romo", "Mauricio Juárez", "María López", "Ernesto Uc", "Carlos Balam", "Jorge Pedrozo", "Didier Gamboa"]
    
    # Array for subjects
    CREATE ARRAY subjects = ["Discrete Mathematics", "Programming", "English II", "Differential Calculus", "Probability and Statistics", "Computer and Server Architecture", "Socio-Emotional Skills and Conflict Management"]
    
    CREATE MATRIX notes[8][7]

    # Login System
    SET logged_in = FALSE
    SET current_role = ""
    SET current_name = ""
    SET user_index = 0
    
    WHILE logged_in is FALSE DO
        PRINT "User: "
        INPUT input_user
        PRINT "Password: "
        INPUT input_pass
        
        SET user_index = 0
        
        # Search for user and validate password
        FOR i = 1 TO total_users DO
            IF usernames[i] EQUALS input_user AND passwords[i] EQUALS input_pass THEN
                SET user_index = i
                SET logged_in = TRUE
                SET current_role = roles[i]
                SET current_name = names[i]
            END IF
        END FOR
        
        IF logged_in is TRUE THEN
            PRINT "Welcome!, " + current_name + " (" + current_role + ")"
        ELSE
            PRINT "Wrong username/password"
        END IF
    END WHILE

    // ==========================================
    // 3. Role Menus
    // ==========================================
    IF current_role EQUALS "student" THEN
        PRINT "========================================="
        PRINT "  School Report"
        PRINT "========================================="
        
        SET approved_list = ""
        SET pending_list = ""
        
        FOR j = 1 TO total_subjects DO
            SET current_grade = notes[user_index][j]
            PRINT subjects[j] + ": " + current_grade
            
            IF current_grade >= 8.0 THEN
                APPEND subjects[j] TO approved_list
            ELSE
                APPEND subjects[j] TO pending_list
            END IF
        END FOR
        
        PRINT "Approved: {" + approved_list + "}"
        PRINT "Pending : {" + pending_list + "}"

    ELSE IF current_role EQUALS "professor" THEN
        PRINT "========================================="
        PRINT "  Students"
        PRINT "========================================="
        
        // Print all students
        FOR i = 1 TO total_users DO
            IF roles[i] EQUALS "student" THEN
                PRINT "User: " + usernames[i] + " | Student: " + names[i]
            END IF
        END FOR
        
        PRINT "Write other thing to exit"
        PRINT "Student to grade (username): "
        INPUT target_user
        
        // Find target user index to evaluate loop condition
        SET target_index = 0
        FOR i = 1 TO total_users DO
            IF usernames[i] EQUALS target_user AND roles[i] EQUALS "student" THEN
                SET target_index = i
            END IF
        END FOR
        
        // Loop while the target_index is valid (student was found)
        WHILE target_index > 0 DO
            PRINT "========================================="
            PRINT "  Subjects"
            PRINT "========================================="
            FOR j = 1 TO total_subjects DO
                PRINT subjects[j]
            END FOR
            
            PRINT "Subject to grade: "
            INPUT target_subject
            
            // Find subject index
            SET subj_index = 0
            FOR j = 1 TO total_subjects DO
                IF subjects[j] EQUALS target_subject THEN
                    SET subj_index = j
                END IF
            END FOR
            
            IF subj_index > 0 THEN
                PRINT "New grade: "
                INPUT new_grade
                
                PRINT "Do you confirm (yes/no)?"
                PRINT target_subject + ": " + notes[target_index][subj_index] + " ==> " + new_grade
                INPUT confirm_change
                
                IF lowercase(confirm_change) EQUALS "yes" THEN
                    SET notes[target_index][subj_index] = new_grade
                    PRINT "Grade updated!"
                    
                    // Print updated grades for that student
                    PRINT "{"
                    FOR j = 1 TO total_subjects DO
                        PRINT subjects[j] + ": " + notes[target_index][j]
                    END FOR
                    PRINT "}"
                END IF
            END IF
            
            PRINT "Write other thing to exit"
            PRINT "Student to grade (username): "
            INPUT target_user
            
            // Re-evaluate target user index for the next loop iteration
            SET target_index = 0
            FOR i = 1 TO total_users DO
                IF usernames[i] EQUALS target_user AND roles[i] EQUALS "student" THEN
                    SET target_index = i
                END IF
            END FOR
        END WHILE

    ELSE IF current_role EQUALS "coordinator" THEN
        PRINT "========================================="
        PRINT "  Professors"
        PRINT "========================================="
        FOR i = 1 TO total_users DO
            IF roles[i] EQUALS "professor" THEN
                PRINT "User: " + usernames[i] + " | Professor: " + names[i]
            END IF
        END FOR
        
        PRINT "========================================="
        PRINT "  Students"
        PRINT "========================================="
        FOR i = 1 TO total_users DO
            IF roles[i] EQUALS "student" THEN
                PRINT "User: " + usernames[i] + " | Student: " + names[i]
            END IF
        END FOR
        
        PRINT "========================================="
        PRINT "  Records"
        PRINT "========================================="
        
        // Print Table Header (Loop to print usernames in one line)
        PRINT_SAME_LINE "SUBJECTS      | "
        FOR i = 1 TO total_users DO
            IF roles[i] EQUALS "student" THEN
                PRINT_SAME_LINE usernames[i] + " | "
            END IF
        END FOR
        PRINT_NEW_LINE()
        
        // Print Table Rows (Loop subjects, then loop students' grades)
        FOR j = 1 TO total_subjects DO
            PRINT_SAME_LINE subjects[j] + " | "
            
            FOR i = 1 TO total_users DO
                IF roles[i] EQUALS "student" THEN
                    PRINT_SAME_LINE notes[i][j] + " | "
                END IF
            END FOR
            
            PRINT_NEW_LINE()
        END FOR

    END IF

END PROGRAM
